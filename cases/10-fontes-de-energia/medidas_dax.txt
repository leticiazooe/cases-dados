Geração Líquida MWh = SUM(fato_geracao_mensal[geracao_liquida_mwh])
Capacidade Instalada MW = SUM(dim_usina[capacidade_mw])
Geração Renovável MWh = CALCULATE([Geração Líquida MWh], dim_fonte_energia[classificacao] = "Renovável")
Participação Renovável % = DIVIDE([Geração Renovável MWh], [Geração Líquida MWh])
Fator de Capacidade % = AVERAGE(fato_geracao_mensal[fator_capacidade])
Disponibilidade % = AVERAGE(fato_geracao_mensal[disponibilidade])
Custo Total R$ = SUM(fato_custo[valor_rs])
Custo Médio R$/MWh = DIVIDE([Custo Total R$], [Geração Líquida MWh])
Emissões tCO2e = SUM(fato_emissao[emissoes_tco2e])
Intensidade de Carbono = DIVIDE([Emissões tCO2e], [Geração Líquida MWh])
Energia Cortada MWh = SUM(fato_geracao_mensal[corte_mwh])
Energia Não Gerada MWh = SUM(fato_indisponibilidade[energia_nao_gerada_mwh])
Receita Contratada R$ = SUM(fato_contrato[receita_contratada_rs])
Desvio CAPEX R$ = SUM(fato_investimento[realizado_rs]) - SUM(fato_investimento[orcamento_rs])
