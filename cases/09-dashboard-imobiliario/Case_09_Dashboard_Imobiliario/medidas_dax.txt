Total Imóveis = DISTINCTCOUNT(fato_imovel[imovel_id])
Valor do Portfólio = SUM(fato_imovel[valor_estimado])
Área Total = SUM(fato_imovel[area_m2])
Preço Médio m² = DIVIDE([Valor do Portfólio], [Área Total])
Receita Mensal Aluguel = CALCULATE(SUM(fato_contrato_aluguel[valor_aluguel]), fato_contrato_aluguel[status] = "Ativo")
Yield Bruto Anual = DIVIDE(12 * SUM(fato_imovel[aluguel_estimado]), SUM(fato_imovel[valor_estimado]))
Imóveis Vagos = CALCULATE(DISTINCTCOUNT(fato_vacancia[imovel_id]), ISBLANK(fato_vacancia[data_fim]))
Taxa de Vacância = DIVIDE([Imóveis Vagos], [Total Imóveis])
Dias Médios de Vacância = AVERAGE(fato_vacancia[dias_vago])
Desconto Médio Venda = AVERAGE(fato_transacao[desconto_pct])
Conversão de Anúncios = DIVIDE(CALCULATE(COUNTROWS(fato_anuncio), fato_anuncio[status] = "Convertido"), COUNTROWS(fato_anuncio))
