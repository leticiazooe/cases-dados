Total Equipamentos = DISTINCTCOUNT(Dim_Equipamento[id_equipamento])
Equipamentos Ativos = CALCULATE([Total Equipamentos], Dim_Equipamento[status_equipamento] = "Ativo")
Total OS = COUNTROWS(Ordens_Servico)
OS Pendentes = CALCULATE([Total OS], Ordens_Servico[status_os] IN {"Planejada","Em execução","Aguardando peça"})
Custo Manutenção = SUM(Ordens_Servico[custo_mao_obra]) + SUM(Ordens_Servico[custo_pecas]) + SUM(Ordens_Servico[custo_terceiros])
Horas Paradas = DIVIDE(SUM(Fato_Paradas[duracao_minutos]),60,0)
MTTR Horas = DIVIDE(AVERAGE(Fato_Paradas[duracao_minutos]),60,0)
Total Falhas = COUNTROWS(Fato_Falhas)
Disponibilidade = DIVIDE(SUM(Fato_Producao[horas_disponiveis]),SUM(Fato_Producao[horas_programadas]),0)
Cumprimento Preventivo = 1 - DIVIDE(CALCULATE(COUNTROWS(Planos_Preventivos),Planos_Preventivos[status_plano]="Vencido"),COUNTROWS(Planos_Preventivos),0)
Taxa Retrabalho = DIVIDE(CALCULATE([Total OS],Ordens_Servico[retrabalho]=1),CALCULATE([Total OS],Ordens_Servico[status_os] IN {"Concluída","Encerrada"}),0)
Custo por Unidade Produzida = DIVIDE([Custo Manutenção],SUM(Fato_Producao[quantidade_produzida]),0)
