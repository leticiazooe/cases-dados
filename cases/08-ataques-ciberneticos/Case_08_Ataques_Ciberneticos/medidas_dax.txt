Alertas Críticos = CALCULATE([Total Alertas], fato_alerta[severidade] = "Crítica")
Taxa Falso Positivo = AVERAGE(fato_alerta[falso_positivo])
Total Incidentes = COUNTROWS(fato_incidente)
Incidentes Abertos = CALCULATE([Total Incidentes], fato_incidente[status] <> "Encerrado")
MTTD Médio Min = AVERAGE(fato_incidente[mttd_min])
MTTR Contenção Min = AVERAGE(fato_incidente[mttr_contencao_min])
Vulnerabilidades Críticas Abertas = CALCULATE(COUNTROWS(fato_vulnerabilidade), fato_vulnerabilidade[severidade] = "Crítica", fato_vulnerabilidade[status] <> "Corrigida")

Taxa Reporte Phishing = AVERAGE(fato_phishing[reportou])
