-- Medidas DAX sugeridas (ajuste os nomes conforme o modelo do Power BI)
Total Projetos = DISTINCTCOUNT(fato_projeto[projeto_id])
Orçamento Aprovado = SUM(fato_projeto[orcamento_aprovado])
Custo Real = SUM(fato_custo[valor_real])
Desvio de Custo = [Custo Real] - [Orçamento Aprovado]
Desvio de Custo % = DIVIDE([Desvio de Custo], [Orçamento Aprovado])
Progresso Médio % = DIVIDE(AVERAGE(fato_projeto[progresso_pct]), 100)
Projetos Atrasados = CALCULATE([Total Projetos], fato_projeto[status] = "Em atraso")
Riscos Altos Abertos = CALCULATE(COUNTROWS(fato_risco), fato_risco[score] >= 15, fato_risco[status] IN {"Aberto","Em mitigação"})
Exposição a Riscos = CALCULATE(SUM(fato_risco[score]), fato_risco[status] IN {"Aberto","Em mitigação"})
Horas Realizadas = SUM(fato_apontamento_horas[horas_regulares]) + SUM(fato_apontamento_horas[horas_extras])
Horas Extras % = DIVIDE(SUM(fato_apontamento_horas[horas_extras]), [Horas Realizadas])
Impacto Mudanças R$ = CALCULATE(SUM(fato_mudanca[impacto_custo]), fato_mudanca[status] IN {"Aprovada","Implementada"})
Impacto Mudanças Dias = CALCULATE(SUM(fato_mudanca[impacto_prazo_dias]), fato_mudanca[status] IN {"Aprovada","Implementada"})
