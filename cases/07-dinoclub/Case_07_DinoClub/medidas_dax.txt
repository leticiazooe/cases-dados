Membros Ativos = CALCULATE(DISTINCTCOUNT(fato_membro[membro_id]), fato_membro[status] = "Ativo")
MRR = SUMX(FILTER(fato_membro, fato_membro[status] = "Ativo"), RELATED(dim_plano[valor_mensal]))
Visitantes = SUM(fato_visita[qtd_pessoas])
Receita Visitas = SUM(fato_visita[valor_pago])
Receita Loja = SUM(fato_venda[valor_liquido])
Ticket Médio Loja = DIVIDE([Receita Loja], DISTINCTCOUNT(fato_venda[venda_id]))
Ocupação Eventos % = DIVIDE(SUM(fato_ingresso_evento[quantidade]), SUM(dim_evento[capacidade]))
CTR = DIVIDE(SUM(fato_interacao_campanha[clique]), SUM(fato_interacao_campanha[impressoes]))
Conversão Marketing % = DIVIDE(SUM(fato_interacao_campanha[conversao]), SUM(fato_interacao_campanha[clique]))
ROI Marketing = DIVIDE(SUM(fato_interacao_campanha[receita_atribuida]) - SUM(dim_campanha[investimento]), SUM(dim_campanha[investimento]))
NPS = 100 * DIVIDE(CALCULATE(COUNTROWS(fato_avaliacao), fato_avaliacao[nps] >= 9), COUNTROWS(fato_avaliacao)) - 100 * DIVIDE(CALCULATE(COUNTROWS(fato_avaliacao), fato_avaliacao[nps] <= 6), COUNTROWS(fato_avaliacao))
